local function createToggle(page, text, callback)
    local frame = Instance.new("Frame")
    frame.Size = UDim2.new(1, 0, 0, 35)
    frame.BackgroundTransparency = 1
    frame.Parent = page
    
    local lbl = Instance.new("TextLabel")
    lbl.Size = UDim2.new(1, -65, 1, 0)
    lbl.BackgroundTransparency = 1
    lbl.Text = text
    lbl.TextColor3 = Color3.fromRGB(230, 230, 230)
    lbl.TextSize = 14
    lbl.Font = Enum.Font.SourceSans
    lbl.TextXAlignment = Enum.TextXAlignment.Left
    lbl.Parent = frame
    
    local btn = Instance.new("TextButton")
    btn.Size = UDim2.new(0, 55, 0, 22)
    btn.Position = UDim2.new(1, -55, 0.5, -11)
    btn.BackgroundColor3 = Color3.fromRGB(200, 50, 50)
    btn.Text = "ВЫКЛ"
    btn.TextColor3 = Color3.fromRGB(255, 255, 255)
    btn.TextSize = 11
    btn.Font = Enum.Font.SourceSansBold
    btn.Parent = frame
    Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 5)
    
    local state = false
    btn.MouseButton1Click:Connect(function()
        if _G.ScriptPaused then return end
        state = not state
        btn.Text = state and "ВКЛ" or "ВЫКЛ"
        btn.BackgroundColor3 = state and Color3.fromRGB(50, 180, 50) or Color3.fromRGB(200, 50, 50)
        callback(state)
    end)

    -- программная смена состояния (без вызова callback), чтобы тумблеры могли выключать друг друга
    return {
        Set = function(value)
            state = value and true or false
            btn.Text = state and "ВКЛ" or "ВЫКЛ"
            btn.BackgroundColor3 = state and Color3.fromRGB(50, 180, 50) or Color3.fromRGB(200, 50, 50)
        end,
        Get = function() return state end,
    }
end

local function findTargetPlayer(str)
    if str == "" then return nil end
    for _, p in pairs(game.Players:GetPlayers()) do
        if p ~= Player and (p.Name:lower():sub(1, #str) == str:lower() or p.DisplayName:lower():sub(1, #str) == str:lower()) then return p end
    end
    return nil
end

local function updateMovement()
    if _G.ScriptPaused or _G.ScriptDestroyed or invisActive then return end
    local char = Player.Character or Player.CharacterAdded:Wait()
    local hum = char:FindFirstChildOfClass("Humanoid")
    if hum then hum.WalkSpeed = walkSpeedValue; hum.JumpPower = jumpPowerValue; hum.UseJumpPower = true end
end
Player.CharacterAdded:Connect(function() task.wait(1) if not _G.ScriptDestroyed then updateMovement() end end)

-- Постоянное удержание заданной скорости и прыжка, чтобы игра не сбрасывала их обратно на 16/50

    if _G.ScriptDestroyed or _G.ScriptPaused then return end
    local char = Player.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if hum then
        if walkSpeedValue ~= 16 and hum.WalkSpeed ~= walkSpeedValue then
            hum.WalkSpeed = walkSpeedValue
        end
        if jumpPowerValue ~= 50 then
            hum.UseJumpPower = true
            if hum.JumpPower ~= jumpPowerValue then
                hum.JumpPower = jumpPowerValue
            end
        end
    end
end)

    local moveDir = hum and hum.MoveDirection or Vector3.new(0, 0, 0)
    if moveDir.Magnitude == 0 and (flying or godmodeEnabled or invisActive) then
        local dir = Vector3.new(0, 0, 0)
        local UIS = game:GetService("UserInputService")
        if UIS:IsKeyDown(Enum.KeyCode.W) then dir = dir + camera.CFrame.LookVector end
        if UIS:IsKeyDown(Enum.KeyCode.S) then dir = dir - camera.CFrame.LookVector end
        if UIS:IsKeyDown(Enum.KeyCode.A) then dir = dir - camera.CFrame.RightVector end
        if UIS:IsKeyDown(Enum.KeyCode.D) then dir = dir + camera.CFrame.RightVector end
        if dir.Magnitude > 0 then return Vector3.new(dir.X, 0, dir.Z).Unit end
    end
    return moveDir
end

local phantomModeType = "none"
local phantomRunning = false
local phantomMiniBtn = nil
local phantomSafeOffset = Vector3.new(3500, 1500, 3500)
local isCastingSkill = false
local castReturnTimer = 0

local function destroyPhantomObjects()
    if godAirPlatform then pcall(function() godAirPlatform:Destroy() end) godAirPlatform = nil end
    if ghostSimulationCube then pcall(function() ghostSimulationCube:Destroy() end) ghostSimulationCube = nil end
    if ghostVisualClone then pcall(function() ghostVisualClone:Destroy() end) ghostVisualClone = nil end
    if godGroundCube then pcall(function() godGroundCube:Destroy() end) godGroundCube = nil end
    if godVisualClone then pcall(function() godVisualClone:Destroy() end) godVisualClone = nil end
    isCastingSkill = false
    castReturnTimer = 0
end

local function triggerQuantumSkillCast(holdDuration)
    if not phantomRunning or not ghostSimulationCube then return end
    local char = Player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end

    castReturnTimer = tick() + (holdDuration or 0.25)
    if isCastingSkill then return end
    isCastingSkill = true

    task.spawn(function()
        while phantomRunning and ghostSimulationCube and hrp and tick() < castReturnTimer and not _G.ScriptDestroyed do
            hrp.Anchored = false
            hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
            hrp.CFrame = ghostSimulationCube.CFrame
            RunService.Heartbeat:Wait()
        end
        if phantomRunning and ghostSimulationCube and hrp then
            hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
            hrp.CFrame = CFrame.new(ghostSimulationCube.Position + phantomSafeOffset + Vector3.new(0, 4, 0))
            hrp.Anchored = true
        end
        isCastingSkill = false
    end)
end

local function stopPhantomSystem(keepMiniBtn)
    phantomRunning = false
    local char = Player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildOfClass("Humanoid")

    if hrp then
        hrp.Anchored = false
        hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
        if ghostSimulationCube then
            hrp.CFrame = CFrame.new(ghostSimulationCube.Position + Vector3.new(0, 3, 0))

local function startPhantomSystem(mode)
    stopPhantomSystem(true)
    local char = Player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hrp or not hum then return end

    phantomModeType = mode
    phantomRunning = true
    local startCFrame = hrp.CFrame
    local isInvis = (mode == "invis")

    godAirPlatform = Instance.new("Part")
    godAirPlatform.Name = "Phantom_AirPlatform"
    godAirPlatform.Size = Vector3.new(30, 2, 30)
    godAirPlatform.Position = startCFrame.Position + phantomSafeOffset
    godAirPlatform.Anchored = true
    godAirPlatform.Transparency = 1
    godAirPlatform.CanCollide = true
    godAirPlatform.Parent = workspace

    ghostSimulationCube = Instance.new("Part")
    ghostSimulationCube.Name = "Phantom_SimCube"
    ghostSimulationCube.Size = Vector3.new(2, 2, 2)
    ghostSimulationCube.CFrame = startCFrame
    ghostSimulationCube.Anchored = true
    ghostSimulationCube.Transparency = 1
    ghostSimulationCube.CanCollide = false
    ghostSimulationCube.Parent = workspace
    godGroundCube = ghostSimulationCube

    char.Archivable = true
    ghostVisualClone = char:Clone()
    ghostVisualClone.Name = "Phantom_Visual"
    ghostVisualClone.Parent = workspace
    godVisualClone = ghostVisualClone

    local cloneRoot = ghostVisualClone:FindFirstChild("HumanoidRootPart")
    if cloneRoot then
        ghostVisualClone.PrimaryPart = cloneRoot
    end

    local origTransparency = {}
    for _, origPart in pairs(char:GetDescendants()) do
        if origPart:IsA("BasePart") then
            origTransparency[origPart.Name] = origPart.Transparency
        end
    end

    for _, p in pairs(ghostVisualClone:GetDescendants()) do
        if p:IsA("BasePart") then
            p.CanCollide = false
            p.Anchored = true
            local wasHidden = (origTransparency[p.Name] and origTransparency[p.Name] >= 0.9)
            if p.Name == "HumanoidRootPart" or p.Name:lower():find("hitbox") or p.Name:lower():find("root") or wasHidden then
                p.Transparency = 1
            else
                p.Transparency = isInvis and 0.65 or 0
            end
        elseif p:IsA("Decal") or p:IsA("Texture") then
            p.Transparency = isInvis and 0.5 or 0
        elseif p:IsA("Script") or p:IsA("LocalScript") then
            p:Destroy()
        end
    end
    local cloneHum = ghostVisualClone:FindFirstChildOfClass("Humanoid")
    if cloneHum then cloneHum.DisplayDistanceType = Enum.HumanoidDisplayDistanceType.None end

    if cloneRoot then
        ghostVisualClone:SetPrimaryPartCFrame(startCFrame)
    end

    local lowestY = startCFrame.Position.Y - 3
    for _, part in pairs(ghostVisualClone:GetDescendants()) do
        if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then
            local partBottom = part.Position.Y - (part.Size.Y * 0.5)
            if partBottom < lowestY then
                lowestY = partBottom
            end
        end
    end
    local hipOffset = math.max(startCFrame.Position.Y - lowestY, 2.5) + 0.05

    hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
    hrp.CFrame = CFrame.new(startCFrame.Position + phantomSafeOffset + Vector3.new(0, 4, 0))
    hrp.Anchored = true

    local camTarget = ghostVisualClone:FindFirstChild("Head") or cloneRoot or ghostSimulationCube
    workspace.CurrentCamera.CameraSubject = camTarget

    local conns = {}
    local function hookTool(tool)
        if tool:IsA("Tool") then
            table.insert(conns, tool.Activated:Connect(function()
                if phantomRunning then triggerQuantumSkillCast(0.28) end
            end))
        end
    end
    for _, t in pairs(char:GetChildren()) do hookTool(t) end
    table.insert(conns, char.ChildAdded:Connect(hookTool))

    table.insert(conns, workspace.DescendantAdded:Connect(function(obj)
        if not phantomRunning or not ghostSimulationCube then return end
        if obj:IsA("BasePart") then
            local n = obj.Name:lower()
            local isMyAttack = obj:IsDescendantOf(char)
                or ((n:find("hitbox") or n:find("slash") or n:find("punch") or n:find("dmg") or n:find("damage"))
                    and (obj.Position - hrp.Position).Magnitude < 25)
            if isMyAttack then
                triggerQuantumSkillCast(0.28)
            end
        end
    end))

    local UIS = game:GetService("UserInputService")
    table.insert(conns, UIS.InputBegan:Connect(function(input, gp)
        if not phantomRunning or gp then return end
        if input.KeyCode == Enum.KeyCode.One or input.KeyCode == Enum.KeyCode.Two or input.KeyCode == Enum.KeyCode.Three or input.KeyCode == Enum.KeyCode.Four then
            triggerQuantumSkillCast(0.28)
        end
    end))

    task.spawn(function()
        local lastTick = tick()
        local verticalVel = 0

        while phantomRunning and ghostSimulationCube and char and hrp and hum and hum.Health > 0 and not _G.ScriptDestroyed do
            local dt = tick() - lastTick
            lastTick = tick()

            if not _G.ScriptPaused then
                if not spectatingPlayer and not freecamEnabled and camTarget and camTarget.Parent then
                    workspace.CurrentCamera.CameraSubject = camTarget
                end

                local moveDir = getCustomMoveDirection(hum)
                local currentPos = ghostSimulationCube.Position

                if moveDir.Magnitude > 0 then
                    local flatDir = Vector3.new(moveDir.X, 0, moveDir.Z).Unit
                    local step = flatDir * (walkSpeedValue * dt)
                    local ray = Ray.new(currentPos, flatDir * (step.Magnitude + 2))
                    local hitPart = workspace:FindPartOnRayWithIgnoreList(ray, {char, ghostSimulationCube, ghostVisualClone, godAirPlatform})
                    if not hitPart or not hitPart.CanCollide or noclip then
                        currentPos = currentPos + step
                    end
                end

                if hum.Jump and verticalVel == 0 then
                    verticalVel = jumpPowerValue
                end

                local downRay = Ray.new(currentPos + Vector3.new(0, 2, 0), Vector3.new(0, -(hipOffset + 3.0), 0))
                local floorPart, floorPos = workspace:FindPartOnRayWithIgnoreList(downRay, {char, ghostSimulationCube, ghostVisualClone, godAirPlatform})

                if floorPart and floorPart.CanCollide and verticalVel <= 0 and not noclip then
                    currentPos = Vector3.new(currentPos.X, floorPos.Y + hipOffset, currentPos.Z)
                    verticalVel = 0
                else
                    verticalVel = verticalVel - (196.2 * dt)
                    currentPos = currentPos + Vector3.new(0, verticalVel * dt, 0)
                    if currentPos.Y < -300 then
                        currentPos = Vector3.new(currentPos.X, 50, currentPos.Z)
                        verticalVel = 0
                    end
                end

                local lookVec = moveDir.Magnitude > 0 and Vector3.new(moveDir.X, 0, moveDir.Z) or ghostSimulationCube.CFrame.LookVector
                if lookVec.Magnitude > 0 then
                    ghostSimulationCube.CFrame = CFrame.new(currentPos, currentPos + lookVec)
                else
                    ghostSimulationCube.CFrame = CFrame.new(currentPos)
                end

                if ghostVisualClone and ghostVisualClone.PrimaryPart then
                    ghostVisualClone:SetPrimaryPartCFrame(ghostSimulationCube.CFrame)
                end

                if not isCastingSkill and godAirPlatform then
                    godAirPlatform.Position = currentPos + phantomSafeOffset
                    hrp.CFrame = CFrame.new(currentPos + phantomSafeOffset + Vector3.new(0, 4, 0))
                end
            end
            RunService.RenderStepped:Wait()
        end

        for _, c in pairs(conns) do pcall(function() c:Disconnect() end) end
        if phantomRunning then stopPhantomSystem(false) end
    end)
end

-- ТУМБЛЕР: ВЫКЛЮЧИТЬ ВСЕ ЗВУКИ КРОМЕ ЛИЧНОГО РАДИО (🔇)
do
    local muteAll = false
    local originalVolumes = setmetatable({}, {__mode = "k"})
    local muteConns = {}

    local function isMyRadio(snd)
        return snd.Name == "PromptPersonalRadio" or snd == customRadioSound
    end

    local function muteSound(obj)
        if not obj:IsA("Sound") or isMyRadio(obj) then return end
        if originalVolumes[obj] == nil then originalVolumes[obj] = obj.Volume end
        pcall(function() obj.Volume = 0 end)
        -- если игра сама поднимет громкость, сразу возвращаем в 0
        table.insert(muteConns, obj:GetPropertyChangedSignal("Volume"):Connect(function()
            if muteAll and obj.Volume ~= 0 and not isMyRadio(obj) then
                pcall(function() obj.Volume = 0 end)
            end
        end))
    end

    createToggle(mainPage, "🔇 Выключить все звуки (кроме радио)", function(Value)
        muteAll = Value
        for _, c in ipairs(muteConns) do pcall(function() c:Disconnect() end) end
        muteConns = {}

        if muteAll then
            for _, d in ipairs(game:GetDescendants()) do muteSound(d) end
            table.insert(muteConns, game.DescendantAdded:Connect(function(d)
                if muteAll then muteSound(d) end
            end))
        else
            for snd, vol in pairs(originalVolumes) do
                if snd.Parent then pcall(function() snd.Volume = vol end) end
            end
            originalVolumes = setmetatable({}, {__mode = "k"})
        end
    end)
end

local function stopFlyPhysics()
    flyActiveState = false
    keysPressed = {w = false, a = false, s = false, d = false}
    if flyVelocity then pcall(function() flyVelocity:Destroy() end) flyVelocity = nil end
    if flyGyro then pcall(function() flyGyro:Destroy() end) flyGyro = nil end

    local char = Player.Character
    if char then
        local hrp = char:FindFirstChild("HumanoidRootPart")
        local hum = char:FindFirstChildOfClass("Humanoid")

        if hrp then
            local look = hrp.CFrame.LookVector
            local flatLook = Vector3.new(look.X, 0, look.Z)
            if flatLook.Magnitude > 0.01 then
                hrp.CFrame = CFrame.new(hrp.Position + Vector3.new(0, 1.2, 0), hrp.Position + Vector3.new(0, 1.2, 0) + flatLook.Unit)
            else
                hrp.CFrame = CFrame.new(hrp.Position + Vector3.new(0, 1.2, 0))
            end
            hrp.AssemblyLinearVelocity = Vector3.new(0, 2, 0)
            hrp.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
            for _, obj in pairs(hrp:GetChildren()) do
                if obj:IsA("BodyVelocity") or obj:IsA("BodyGyro") then obj:Destroy() end
            end
        end

        if hum then
            local animator = hum:FindFirstChildOfClass("Animator")
            if animator then
                for _, track in pairs(animator:GetPlayingAnimationTracks()) do track.Speed = 1 end
            end
            hum.PlatformStand = false
            hum.Sit = false
            hum.AutoRotate = true
            hum:ChangeState(Enum.HumanoidStateType.GettingUp)
            task.delay(0.05, function()
                if hum and hum.Parent and not flyActiveState then
                    hum.PlatformStand = false
                    hum.AutoRotate = true
                    hum:ChangeState(Enum.HumanoidStateType.Running)
                    hum:Move(Vector3.new(0, 0, 0))
                end
            end)
        end
    end
end

local function startFlyPhysics()
    if flyVelocity then pcall(function() flyVelocity:Destroy() end) flyVelocity = nil end
    if flyGyro then pcall(function() flyGyro:Destroy() end) flyGyro = nil end
    
    local char = Player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hrp or not hum then return end
    
    flyActiveState = true
    flyVelocity = Instance.new("BodyVelocity")
    flyVelocity.MaxForce = Vector3.new(9e9, 9e9, 9e9)
    flyVelocity.P = 12500
    flyVelocity.Velocity = Vector3.new(0, 0, 0)
    flyVelocity.Parent = hrp
    
    flyGyro = Instance.new("BodyGyro")
    flyGyro.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
    flyGyro.P = 15000
    flyGyro.CFrame = hrp.CFrame
    flyGyro.Parent = hrp
    
    task.spawn(function()
        local camera = workspace.CurrentCamera
        local lastFlyTick = tick()
        while flyActiveState and flying and char and hrp and hum and not _G.ScriptDestroyed do
            local dt = tick() - lastFlyTick
            lastFlyTick = tick()

            if not _G.ScriptPaused then
                hum.PlatformStand = true
                
                if noclip then
                    for _, part in pairs(char:GetDescendants()) do
                        if part:IsA("BasePart") then part.CanCollide = false end
                    end
                end
                
                local animator = hum:FindFirstChildOfClass("Animator")
                if animator then
                    for _, track in pairs(animator:GetPlayingAnimationTracks()) do 
                        local n = track.Animation.Name:lower()
                        if n:find("fall") or n:find("jump") or n:find("run") then
                            track.Speed = 0 
                        end
                    end
                end
                
                if flyGyro then flyGyro.CFrame = camera.CFrame end
                
                local flyVector = Vector3.new(0, 0, 0)
                local camLook = camera.CFrame.LookVector
                local camRight = camera.CFrame.RightVector
                local UIS = game:GetService("UserInputService")
                
                if keysPressed.w or UIS:IsKeyDown(Enum.KeyCode.W) then flyVector = flyVector + camLook end 
                if keysPressed.s or UIS:IsKeyDown(Enum.KeyCode.S) then flyVector = flyVector - camLook end 
                if keysPressed.a or UIS:IsKeyDown(Enum.KeyCode.A) then flyVector = flyVector - camRight end 
                if keysPressed.d or UIS:IsKeyDown(Enum.KeyCode.D) then flyVector = flyVector + camRight end
                
                -- Прямой сдвиг CFrame + обнуление скорости (пробивает кастомный контроллер в SCP-3008)
                hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
                if flyVector.Magnitude > 0 then
                    local dir = flyVector.Unit
                    if flyVelocity then flyVelocity.Velocity = dir * flySpeedValue end
                    hrp.CFrame = CFrame.new(hrp.Position + dir * (flySpeedValue * dt)) * (camera.CFrame - camera.CFrame.Position)
                else
                    if flyVelocity then flyVelocity.Velocity = Vector3.new(0, 0, 0) end
                    hrp.CFrame = CFrame.new(hrp.Position) * (camera.CFrame - camera.CFrame.Position)
                end
            else
                if flyVelocity then flyVelocity.Velocity = Vector3.new(0, 0, 0) end
            end
            RunService.RenderStepped:Wait()
        end
    end)
end

local function findExactOrPrefixPlayer(query)
    if not query or query == "" then return nil end
    local q = query:lower():gsub("^%s*(.-)%s*$", "%1")
    if q == "" then return nil end

    -- Сначала ищем точное совпадение по нику или DisplayName
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= Player then
            if plr.Name:lower() == q or plr.DisplayName:lower() == q then
                return plr
            end
        end
    end

    -- Затем ищем по началу ника или DisplayName
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= Player then
            if plr.Name:lower():sub(1, #q) == q or plr.DisplayName:lower():sub(1, #q) == q then
                return plr
            end
        end
    end
    return nil
end

local function findNearestEnemyToCenter()
    local cam = workspace.CurrentCamera
    local center = Vector2.new(cam.ViewportSize.X * 0.5, cam.ViewportSize.Y * 0.5)
    local bestPlr = nil
    local bestScore = math.huge

    local myChar = Player.Character
    local myRoot = (phantomRunning and ghostSimulationCube) or (myChar and myChar:FindFirstChild("HumanoidRootPart"))

    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= Player and plr.Character then
            local tHrp = plr.Character:FindFirstChild("HumanoidRootPart")
            local tHum = plr.Character:FindFirstChildOfClass("Humanoid")
            if tHrp and tHum and tHum.Health > 0 then
                local worldDist = myRoot and (myRoot.Position - tHrp.Position).Magnitude or 0
                if worldDist <= 250 then
                    local screenPos, onScreen = cam:WorldToViewportPoint(tHrp.Position)
                    local screenDist = onScreen and (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude or 9999
                    local score = (worldDist <= 20) and worldDist or (screenDist + worldDist * 0.5)
                    if score < bestScore then
                        bestScore = score
                        bestPlr = plr
                    end
                end
            end
        end
    end
    return bestPlr
end

-- ШИФТЛОК: пока он включён, "мышь" для игры = центр экрана (как заблокированная мышь на ПК)
-- Работает через hookmetamethod; без него шифтлок просто остаётся камерой/поворотом как раньше.
task.spawn(function()
    repeat task.wait(0.5) until shiftLockActive or _G.ScriptDestroyed
    if _G.ScriptDestroyed then return end
    if not (hookmetamethod and checkcaller and getnamecallmethod) then
        warn("[Шифтлок] Экзекутор не поддерживает hookmetamethod: выстрелы в центр недоступны")
        return
    end

    local GuiService = game:GetService("GuiService")
    local UIS = game:GetService("UserInputService")
    local mouse = Player:GetMouse()
    local wrap = newcclosure or function(f) return f end

    local function centerPoint()
        local vs = workspace.CurrentCamera.ViewportSize
        return Vector2.new(vs.X * 0.5, vs.Y * 0.5)
    end

    local function centerRay()
        local c = centerPoint()
        return workspace.CurrentCamera:ViewportPointToRay(c.X, c.Y)
    end

    local function centerRaycast()
        local ray = centerRay()
        local params = RaycastParams.new()
        params.FilterType = Enum.RaycastFilterType.Exclude
        local ignore = {}
        if Player.Character then table.insert(ignore, Player.Character) end
        if ghostVisualClone then table.insert(ignore, ghostVisualClone) end
        if ghostSimulationCube then table.insert(ignore, ghostSimulationCube) end
        params.FilterDescendantsInstances = ignore
        local result = workspace:Raycast(ray.Origin, ray.Direction * 1000, params)
        return ray, result
    end

    local function redirectActive()
        return shiftLockActive and not _G.ScriptPaused and not _G.ScriptDestroyed
    end

    local oldIndex
    oldIndex = hookmetamethod(game, "__index", wrap(function(self, key)
        if self == mouse and not checkcaller() and redirectActive() then
            if key == "Hit" then
                local ray, result = centerRaycast()
                return CFrame.new(result and result.Position or (ray.Origin + ray.Direction * 1000))
            elseif key == "Target" then
                local _, result = centerRaycast()
                return result and result.Instance or nil
            elseif key == "X" then
                return centerPoint().X - GuiService:GetGuiInset().X
            elseif key == "Y" then
                return centerPoint().Y - GuiService:GetGuiInset().Y
            elseif key == "UnitRay" then
                return centerRay()
            end
        end
        return oldIndex(self, key)
    end))

    local oldNamecall
    oldNamecall = hookmetamethod(game, "__namecall", wrap(function(self, ...)
        if self == UIS and not checkcaller() and redirectActive() and getnamecallmethod() == "GetMouseLocation" then
            return centerPoint()
        end
        return oldNamecall(self, ...)
    end))
end)

end

local function isDynamicSkillHitbox(obj)
    if not obj:IsA("BasePart") then return false end
    if obj:FindFirstChild("PromptHitboxBox") then return false end
    if Player.Character and obj:IsDescendantOf(Player.Character) then return false end
    if ghostVisualClone and obj:IsDescendantOf(ghostVisualClone) then return false end
    if obj.Name:find("Phantom_") or obj.Name:find("ZoneBoxVisual_") or obj.Name == "Baseplate" then return false end

    local maxAxis = math.max(obj.Size.X, obj.Size.Y, obj.Size.Z)
    if maxAxis < 1.5 or maxAxis > 250 then return false end

    local lowerName = obj.Name:lower()
    local parentName = obj.Parent and obj.Parent.Name:lower() or ""

    local hasTouch = obj:FindFirstChildOfClass("TouchTransmitter") ~= nil
    local keywordMatch = lowerName:find("hitbox") or lowerName:find("damage") or lowerName:find("dmg")
        or lowerName:find("hurt") or lowerName:find("tornado") or lowerName:find("hurricane")
        or lowerName:find("vortex") or lowerName:find("pull") or lowerName:find("wind")
        or lowerName:find("shock") or lowerName:find("blast") or lowerName:find("zone")
        or lowerName:find("aura") or lowerName:find("slash") or lowerName:find("wave")
        or parentName:find("effect") or parentName:find("skill") or parentName:find("ability")
        or parentName:find("projectile") or parentName:find("attack") or parentName:find("vfx")

    local isInvisibleTrigger = (not obj.CanCollide) and (obj.Transparency >= 0.4)

    return hasTouch or keywordMatch or isInvisibleTrigger
end

local playerCooldownData = {}

local function registerPlayerTool(pData, toolObj)
    if not toolObj or not toolObj:IsA("Tool") then return end
    local toolName = toolObj.Name
    local slotIdx = pData.NameToSlot[toolName]

    if not slotIdx then
        if #pData.SlotNames < 9 then
            table.insert(pData.SlotNames, toolName)
            slotIdx = #pData.SlotNames
            pData.NameToSlot[toolName] = slotIdx
        else
            slotIdx = 9
        end
    end

    if not pData.HookedTools[toolObj] then
        pData.HookedTools[toolObj] = true
        table.insert(pData.Connections, toolObj.Activated:Connect(function()
            local cd = toolObj:GetAttribute("Cooldown") or toolObj:GetAttribute("CD") or 4
            if type(cd) ~= "number" or cd <= 0 then cd = 4 end
            pData.Cooldowns[slotIdx] = tick() + cd
        end))
    end
end

local function setupPlayerCdTracker(plr, char, hum)
    local pData = {
        Char = char,
        SlotNames = {},
        NameToSlot = {},
        Cooldowns = {},
        HookedTools = {},
        LastAnimTime = 0,
        Connections = {}
    }
    playerCooldownData[plr] = pData

    table.insert(pData.Connections, char.ChildAdded:Connect(function(child)
        if child:IsA("Tool") then registerPlayerTool(pData, child) end
    end))

    local animator = hum:FindFirstChildOfClass("Animator")
    if animator then
        table.insert(pData.Connections, animator.AnimationPlayed:Connect(function(track)
            if tick() - pData.LastAnimTime < 0.45 then return end
            local animName = track.Animation and track.Animation.Name:lower() or ""
            if animName:find("walk") or animName:find("run") or animName:find("idle") or animName:find("jump") or animName:find("fall") then
                return
            end
            pData.LastAnimTime = tick()

            local heldTool = char:FindFirstChildOfClass("Tool")
            if heldTool then
                registerPlayerTool(pData, heldTool)
                local sIdx = pData.NameToSlot[heldTool.Name]
                if sIdx and (not pData.Cooldowns[sIdx] or pData.Cooldowns[sIdx] < tick()) then
                    local cd = heldTool:GetAttribute("Cooldown") or heldTool:GetAttribute("CD") or 4
                    if type(cd) ~= "number" or cd <= 0 then cd = 4 end
                    pData.Cooldowns[sIdx] = tick() + cd
                end
            end
        end))
    end

    return pData
end

local function evaluateMathExpression(rawText)
    if not rawText or type(rawText) ~= "string" then return nil, nil end

    local cleaned = rawText:gsub("×", "*"):gsub("х", "*"):gsub("Х", "*"):gsub("x", "*"):gsub("X", "*")
    cleaned = cleaned:gsub("÷", "/"):gsub(":", "/")

    local aStr, op, bStr = cleaned:match("(%-?%d+%.?%d*)%s*([%+%-/%*])%s*(%-?%d+%.?%d*)")
    if aStr and op and bStr then
        local a = tonumber(aStr)
        local b = tonumber(bStr)
        if a and b then
            local res = nil
            if op == "+" then res = a + b
            elseif op == "-" then res = a - b
            elseif op == "*" then res = a * b
            elseif op == "/" and b ~= 0 then res = a / b end

            if res ~= nil then
                local formattedRes = (res % 1 == 0) and tostring(math.floor(res)) or string.format("%.2f", res)
                return string.format("%s %s %s", aStr, op, bStr), formattedRes
            end
        end
    end

    local safeExpr = cleaned:gsub("[^%d%+%-%*/%.%(%)%s]", "")
    if safeExpr:find("[%+%-%*/]") and safeExpr:find("%d") then
        local fn = loadstring("return " .. safeExpr)
        if fn then
            local ok, val = pcall(fn)
            if ok and type(val) == "number" and val == val and math.abs(val) < 1e15 then
                local formattedRes = (val % 1 == 0) and tostring(math.floor(val)) or string.format("%.2f", val)
                return safeExpr, formattedRes
            end
        end
    end

    return nil, nil
end

local function startBendyDive(hudBtn)
    local char = Player.Character
    local realHrp = char and char:FindFirstChild("HumanoidRootPart")
    local realHum = char and char:FindFirstChildOfClass("Humanoid")
    if not realHrp or not realHum then return end

    local walkAnim, jumpAnim
    local origAnim = char:FindFirstChild("Animate")
    if origAnim then
        local r = origAnim:FindFirstChild("run") or origAnim:FindFirstChild("walk")
        if r and r:FindFirstChildOfClass("Animation") then
            walkAnim = r:FindFirstChildOfClass("Animation"):Clone()
        end
        local j = origAnim:FindFirstChild("jump")
        if j and j:FindFirstChildOfClass("Animation") then
            jumpAnim = j:FindFirstChildOfClass("Animation"):Clone()
        end
    end

    char.Archivable = true
    local proxy = char:Clone()
    char.Archivable = false
    proxy.Name = "BendySurfacePuppet"

    for _, obj in ipairs(proxy:GetDescendants()) do
        if obj:IsA("LocalScript") or obj:IsA("Script") or obj:IsA("ForceField") then
            obj:Destroy()
        elseif obj:IsA("BasePart") then
            obj.Anchored = false
            obj.Material = Enum.Material.ForceField
            obj.Color = Color3.fromRGB(35, 35, 45)
            obj.Transparency = (obj.Name == "HumanoidRootPart") and 1 or 0.55
            obj.CanCollide = (obj.Name == "HumanoidRootPart" or obj.Name == "Torso" or obj.Name == "UpperTorso" or obj.Name == "LowerTorso")
        elseif obj:IsA("Decal") or obj:IsA("Texture") then
            obj.Transparency = 0.55
        end
    end

    local pHrp = proxy:FindFirstChild("HumanoidRootPart")
    local pHum = proxy:FindFirstChildOfClass("Humanoid")
    if not pHrp or not pHum then
        proxy:Destroy()
        return
    end

    pHum.DisplayDistanceType = Enum.HumanoidDisplayDistanceType.None
    pHum.WalkSpeed = realHum.WalkSpeed
    pHum.JumpPower = realHum.JumpPower
    pHum.UseJumpPower = realHum.UseJumpPower
    pHum.JumpHeight = realHum.JumpHeight
    pHum.AutoRotate = true
    proxy.Parent = workspace
    bendyProxyRig = proxy

    proxy:PivotTo(realHrp.CFrame)
    workspace.CurrentCamera.CameraSubject = pHum

    local animator = pHum:FindFirstChildOfClass("Animator") or Instance.new("Animator", pHum)
    local walkTrack, jumpTrack
    if walkAnim then
        walkAnim.Parent = proxy
        walkTrack = animator:LoadAnimation(walkAnim)
        walkTrack.Looped = true
    end
    if jumpAnim then
        jumpAnim.Parent = proxy
        jumpTrack = animator:LoadAnimation(jumpAnim)
        jumpTrack.Looped = false
    end

    pHum.Running:Connect(function(spd)
        if not walkTrack then return end
        if spd > 1 then
            if not walkTrack.IsPlaying then walkTrack:Play(0.15) end
            walkTrack:AdjustSpeed(spd / 16)
        else
            if walkTrack.IsPlaying then walkTrack:Stop(0.15) end
        end
    end)

    pHum.StateChanged:Connect(function(_, state)
        if not jumpTrack then return end
        if state == Enum.HumanoidStateType.Jumping or state == Enum.HumanoidStateType.Freefall then
            if not jumpTrack.IsPlaying then jumpTrack:Play(0.1) end
        else
            if jumpTrack.IsPlaying then jumpTrack:Stop(0.15) end
        end
    end)

    bendyConnStepped = RunService.Stepped:Connect(function()
        if not bendyDiveActive or not char.Parent then return end
        for _, part in ipairs(char:GetDescendants()) do
            if part:IsA("BasePart") then
                part.CanCollide = false
            end
        end
    end)

    local bob = 0
    bendyConnHeartbeat = RunService.Heartbeat:Connect(function()
        if not bendyDiveActive then return end
        if _G.ScriptDestroyed or not char.Parent or not realHrp.Parent or not proxy.Parent or realHum.Health <= 0 then

            if hudBtn then
                hudBtn.Text = "OFF"
                hudBtn.BackgroundColor3 = Color3.fromRGB(205, 55, 45)
            end
            stopBendyDive(false)
            return
        end

        pHum.WalkSpeed = realHum.WalkSpeed
        pHum:Move(realHum.MoveDirection, false)
        if realHum.Jump then
            pHum.Jump = true
        end

        bob = (bob == 0) and 0.1 or 0
        local voidLimit = workspace.FallenPartsDestroyHeight + 3
        local deepY = math.max(pHrp.Position.Y - 35 + bob, voidLimit)

        realHum:ChangeState(Enum.HumanoidStateType.Physics)
        realHrp.AssemblyLinearVelocity = Vector3.new(0, 0.4, 0)
        realHrp.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
        realHrp.CFrame = CFrame.new(pHrp.Position.X, deepY, pHrp.Position.Z) * (pHrp.CFrame - pHrp.CFrame.Position)
    end)

local function buildBoxVisual(ptA, ptB, boxTitle, ruleStr)
    local sizeX = math.abs(ptA.X - ptB.X) + 10
    local sizeZ = math.abs(ptA.Z - ptB.Z) + 10
    local center = Vector3.new((ptA.X + ptB.X)/2, (ptA.Y + ptB.Y)/2, (ptA.Z + ptB.Z)/2)
    local col = getBoxColorByRule(ruleStr)

    local part = Instance.new("Part")
    part.Name = "ZoneBoxVisual_" .. boxTitle
    part.Size = Vector3.new(sizeX, 20, sizeZ)
    part.Position = center
    part.Anchored = true
    part.CanCollide = false
    part.Transparency = 0.8
    part.Material = Enum.Material.ForceField
    part.Color = col
    part.Parent = workspace

    local sel = Instance.new("SelectionBox", part)
    sel.Name = "BoxSel"
    sel.Adornee = part
    sel.Color3 = col
    sel.LineThickness = 0.08

    local bbg = Instance.new("BillboardGui", part)
    bbg.Name = "BoxTitleGui"
    bbg.AlwaysOnTop = true
    bbg.MaxDistance = 15000
    bbg.Size = UDim2.new(0, 260, 0, 50)
    bbg.StudsOffset = Vector3.new(0, 14, 0)

    local lbl = Instance.new("TextLabel", bbg)
    lbl.Name = "TitleLbl"
    lbl.Size = UDim2.new(1, 0, 1, 0)
    lbl.BackgroundTransparency = 1
    lbl.Text = "📦 " .. boxTitle
    lbl.TextColor3 = col
    lbl.TextStrokeTransparency = 0.2
    lbl.Font = Enum.Font.SourceSansBold
    lbl.TextSize = 20

    task.spawn(function()
        local cam = workspace.CurrentCamera
        while part and part.Parent and not _G.ScriptDestroyed do
            local dist = (cam.CFrame.Position - part.Position).Magnitude
            lbl.TextSize = math.clamp(18 + (dist / 40), 18, 95)
            bbg.Size = UDim2.new(0, math.clamp(240 + (dist / 8), 240, 1200), 0, math.clamp(45 + (dist / 30), 45, 250))
            task.wait(0.2)
        end
    end)

    return part
end

    end
end

local function finalizeNewBox()
    if editingBoxData or not draftPointA or not draftPointB then return end
    local boxName = nameBox.Text:gsub("^%s*(.-)%s*$", "%1")
    if boxName == "" then return end
    if draftMarkerA then draftMarkerA:Destroy() draftMarkerA = nil end
    if draftMarkerB then draftMarkerB:Destroy() draftMarkerB = nil end
    local mA = createZoneMarker(draftPointA, boxName .. " [А]", Color3.fromRGB(255, 220, 60))
    local mB = createZoneMarker(draftPointB, boxName .. " [Б]", Color3.fromRGB(60, 220, 255))
    local vis = buildBoxVisual(draftPointA, draftPointB, boxName, "")
    table.insert(_G.SavedZoneBoxes, {Name = boxName, PointA = draftPointA, PointB = draftPointB, MarkerA = mA, MarkerB = mB, Visual = vis, Rule = "", Active = true})
    draftPointA, draftPointB, nameBox.Text = nil, nil, ""
    updateCreatorUI()
    refreshBoxesList()
end

                    local char = Player.Character
                    local hrp = (invisActive and ghostSimulationCube) or (godmodeEnabled and godGroundCube) or (char and char:FindFirstChild("HumanoidRootPart"))
                    if hrp then
                        for _, bData in ipairs(_G.SavedZoneBoxes) do
                            if bData.Active and bData.PointA and bData.PointB and (bData.Rule == "+" or bData.Rule == "-") then
                                local minX = math.min(bData.PointA.X, bData.PointB.X) - 5
                                local maxX = math.max(bData.PointA.X, bData.PointB.X) + 5
                                local midY = (bData.PointA.Y + bData.PointB.Y) / 2
                                local minY, maxY = midY - 10, midY + 10
                                local minZ = math.min(bData.PointA.Z, bData.PointB.Z) - 5
                                local maxZ = math.max(bData.PointA.Z, bData.PointB.Z) + 5
                                local pos = hrp.Position
                                local isInside = (pos.X >= minX and pos.X <= maxX) and (pos.Y >= minY and pos.Y <= maxY) and (pos.Z >= minZ and pos.Z <= maxZ)

                                if bData.Rule == "+" and not isInside then
                                    local clamped = Vector3.new(math.clamp(pos.X, minX + 1.5, maxX - 1.5), math.clamp(pos.Y, minY + 1.5, maxY - 1.5), math.clamp(pos.Z, minZ + 1.5, maxZ - 1.5))
                                    hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
                                    hrp.CFrame = CFrame.new(clamped) * (hrp.CFrame - hrp.CFrame.Position)
                                elseif bData.Rule == "-" and isInside then
                                    local dL, dR, dT, dB = math.abs(pos.X - minX), math.abs(maxX - pos.X), math.abs(pos.Z - minZ), math.abs(maxZ - pos.Z)
                                    local mD = math.min(dL, dR, dT, dB)
                                    local pushPos = (mD == dL and Vector3.new(minX - 6, pos.Y, pos.Z)) or (mD == dR and Vector3.new(maxX + 6, pos.Y, pos.Z)) or (mD == dT and Vector3.new(pos.X, pos.Y, minZ - 6)) or Vector3.new(pos.X, pos.Y, maxZ + 6)
                                    hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
                                    hrp.CFrame = CFrame.new(pushPos) * (hrp.CFrame - hrp.CFrame.Position)
                                end
                            end
                        end
                    end
                end
                RunService.Heartbeat:Wait()
            end
        end)
    end
end)

local function recalculateOrders()
    orderCounter = 0
    local sorted = {}
    for _, pt in ipairs(savedPoints) do
        if pt.Order and pt.Order > 0 then table.insert(sorted, pt) end
    end
    table.sort(sorted, function(a,b) return a.Order < b.Order end)
    for i, pt in ipairs(sorted) do
        pt.Order = i
        if pt.NumBox then pt.NumBox.Text = tostring(i) end
        orderCounter = i
    end
end

local function moveToDest(hrp, dest, abortFn)
    if isTeleportMode then
        hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
        hrp.CFrame = CFrame.new(dest)
        return
    end
    local wasAnc = hrp.Anchored
    if not dropOnPoint then hrp.Anchored = true end
    while hrp and hrp.Parent and not _G.ScriptDestroyed do
        if abortFn and abortFn() then break end
        local diff = dest - hrp.Position
        local dist = diff.Magnitude
        if dist <= 1.8 then
            hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
            hrp.CFrame = CFrame.new(dest)
            break
        end
        local dt = RunService.Heartbeat:Wait()
        hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
        hrp.CFrame = CFrame.new(hrp.Position + diff.Unit * math.min(dist, ziplineSpeedValue * dt))
    end
    if hrp and hrp.Parent and not invisActive then hrp.Anchored = wasAnc end
end

        repeat
            local ordered = {}
            for _, pt in ipairs(savedPoints) do
                if pt.Order and pt.Order > 0 then table.insert(ordered, pt) end
            end
            table.sort(ordered, function(a, b) return a.Order < b.Order end)
            if #ordered == 0 then break end

            for _, pt in ipairs(ordered) do
                if not ziplineActive or _G.ScriptDestroyed then break end
                local r = (invisActive and ghostSimulationCube) or (godmodeEnabled and godGroundCube) or (Player.Character and Player.Character:FindFirstChild("HumanoidRootPart"))
                if not r then break end
                moveToDest(r, pt.Pos + Vector3.new(0, 3, 0), function() return not ziplineActive end)
                if ziplineDelayValue > 0 and ziplineActive then task.wait(ziplineDelayValue) end
            end
            task.wait(0.05)
        until not isZiplineLoop or not ziplineActive or _G.ScriptDestroyed

        ziplineActive = false
        runZiplineBtn.BackgroundColor3 = Color3.fromRGB(220, 180, 40)
        runZiplineBtn.Text = "🚀 Старт"
    end)
end)

local function bindDeathTracker(char)
    if not char then return end
    local hum = char:WaitForChild("Humanoid", 5)
    local hrp = char:WaitForChild("HumanoidRootPart", 5)
    if hum and hrp then
        hum.Died:Connect(function() if hrp and hrp.Parent then lastDeathPos = hrp.Position end end)
    end
end
if Player.Character then task.spawn(bindDeathTracker, Player.Character) end

Player.CharacterAdded:Connect(function(newChar)
    task.spawn(bindDeathTracker, newChar)
    if not deathZiplineEnabled or _G.ScriptPaused or _G.ScriptDestroyed then return end
    local newHrp = newChar:WaitForChild("HumanoidRootPart", 5)
    if not newHrp then return end
    task.wait(0.45)

    local ordered = {}
    for _, pt in ipairs(savedPoints) do
        if pt.Order and pt.Order > 0 then table.insert(ordered, pt) end
    end
    table.sort(ordered, function(a, b) return a.Order < b.Order end)
    if #ordered == 0 then return end

    local backPos = lastDeathPos or newHrp.Position
    for _, pt in ipairs(ordered) do
        if not deathZiplineEnabled or _G.ScriptDestroyed or not newHrp.Parent then break end
        moveToDest(newHrp, pt.Pos + Vector3.new(0, 3, 0), function() return not deathZiplineEnabled end)
        if ziplineDelayValue > 0 then task.wait(ziplineDelayValue) end
    end
    if deathZiplineEnabled and newHrp.Parent and backPos then
        moveToDest(newHrp, backPos + Vector3.new(0, 3, 0), function() return not deathZiplineEnabled end)
    end
end)
